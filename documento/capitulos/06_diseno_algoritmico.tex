\section{Diseño Algorítmico y Estructuras de Cola}
\label{sec:diseno_algoritmico}

\subsection{Algoritmo de Dijkstra parametrizado}
El Algoritmo~\ref{alg:dijkstra_generico} describe el núcleo común. La única dependencia de la cola es el contrato \texttt{insert}, \texttt{extract\_min} y \texttt{decrease\_key}; el resto del algoritmo no cambia.

\begin{algorithm}[H]
\caption{Dijkstra con cola de prioridad parametrizada}
\label{alg:dijkstra_generico}
\begin{algorithmic}[1]
\REQUIRE $G=(V,E,w)$, origen $s\in V$, cola $Q$
\ENSURE $d:V\to\mathbb{R}_{\ge0}\cup\{\infty\}$, $\pi:V\to V\cup\{\text{NIL}\}$
\FORALL{$v\in V$}
    \STATE $d[v]\leftarrow\infty$; $\pi[v]\leftarrow\text{NIL}$
\ENDFOR
\STATE $d[s]\leftarrow 0$; $Q.\text{insert}(s,0)$; $S\leftarrow\emptyset$
\WHILE{\textbf{not} $Q.\text{is\_empty}()$}
    \STATE $(d_u,u)\leftarrow Q.\text{extract\_min}()$
    \IF{$d_u=\infty$}
        \STATE \textbf{break}
    \ENDIF
    \STATE $S\leftarrow S\cup\{u\}$
    \FORALL{$(u,v)\in E$}
        \IF{$v\notin S$ \textbf{and} $d_u+w(u,v)<d[v]$}
            \STATE $d[v]\leftarrow d_u+w(u,v)$; $\pi[v]\leftarrow u$
            \IF{$v\in Q$}
                \STATE $Q.\text{decrease\_key}(v,d[v])$
            \ELSE
                \STATE $Q.\text{insert}(v,d[v])$
            \ENDIF
        \ENDIF
    \ENDFOR
\ENDWHILE
\RETURN $(d,\pi)$
\end{algorithmic}
\end{algorithm}

\subsection{Invariante y corrección}
\paragraph{Invariante.} Al inicio de cada iteración del ciclo, para todo $u\in S$, $d[u]=\delta(s,u)$; para todo $v\in Q$, $d[v]$ es el costo mínimo de un camino de $s$ a $v$ con vértices intermedios en $S$.

\paragraph{Demostración.} La base es $S=\emptyset$, $d[s]=0$. Supóngase el invariante para $|S|=k$ y sea $u$ el mínimo extraído. Si existiera un camino $P$ de $s$ a $u$ con $w(P)<d[u]$, sea $(x,y)$ la primera arista de $P$ que sale de $S$. Entonces $x\in S$, $y\in Q$ y
\begin{equation}
    w(P)\ge \delta(s,x)+w(x,y)=d[y]\ge d[u],
\end{equation}
pues los pesos son no negativos y $u$ minimiza $Q$. Contradicción. Por tanto $d[u]=\delta(s,u)$. Al terminar, $Q=\emptyset$ o solo quedan vértices inalcanzables, y el invariante cubre toda la componente alcanzable.

\subsection{Heap binario indexado}
El heap binario se almacena en un arreglo contiguo. Para que \texttt{decrease\_key} no requiera una búsqueda lineal, se mantiene un mapa $\text{pos}[v]$ con la posición actual de $v$. Cada intercambio actualiza el arreglo y el mapa:
\begin{equation}
    \text{heap}[i]\leftrightarrow\text{heap}[j],\qquad
    \text{pos}[\text{heap}[i]]\leftarrow i,\qquad
    \text{pos}[\text{heap}[j]]\leftarrow j.
    \label{eq:swap}
\end{equation}
Así, \texttt{decrease\_key} localiza el nodo en $O(1)$ y sube a lo sumo $\lfloor\log_2 |V|\rfloor$ niveles. El costo espacial es $O(|V|)$.

\subsection{Heap 4-ario indexado}
El heap 4-ario reduce la altura a $\lceil\log_4|V|\rceil$. La aritmética de índices es la de la ecuación~\eqref{eq:aritmetica_4ary}. La operación \texttt{sift-down} examina cuatro hijos por nivel; la operación \texttt{sift-up} hace una comparación por nivel. Con $d=4$ fijo, el costo total de Dijkstra es
\begin{equation}
    O\bigl((|E|+4|V|)\log_4|V|\bigr)=O\bigl((|E|+|V|)\log|V|\bigr),
\end{equation}
con una constante que depende de $d$. La ventaja potencial no es asintótica, sino de localidad: los cuatro hijos de un nodo ocupan posiciones contiguas y suelen caber en una misma línea de caché \parencite{lamarca1996cache}. Esa ventaja puede desaparecer si el tamaño es pequeño o si el costo de comparar cuatro hijos domina.

\subsection{Fibonacci heap}
El Fibonacci heap mantiene una lista circular de árboles. \texttt{insert} añade un nodo a la lista en $O(1)$; \texttt{decrease\_key} corta el nodo si viola la propiedad de montículo y propaga el corte en cascada si el padre ya estaba marcado; \texttt{extract\_min} promueve los hijos y consolida raíces de igual grado. Sus cotas amortizadas son $O(1)$ para \texttt{insert} y \texttt{decrease\_key}, y $O(\log |V|)$ para \texttt{extract\_min}. El costo total es $O(|E|+|V|\log|V|)$ \parencite{fredman1987fibonacci}. La cota es mejor en el peor caso asintótico, pero no dice nada sobre las constantes de asignación de objetos, indirección de punteros y fallos de caché.
